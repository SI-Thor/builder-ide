(async () => {
  const start = { x: 37.6, y: 67, z: 44.7 };
  const logTypes = ['oak_log','spruce_log','birch_log','jungle_log','acacia_log','dark_oak_log','mangrove_log','cherry_log'];
  const isLog = (b) => typeof b === 'string' && b.endsWith('_log');

  const findLog = async (radius) => {
    const s = await mc.scan(radius);
    const found = Object.keys(s.nearby || {}).find(isLog);
    if (found) {
      console.log('found ' + found + ' in scan radius ' + radius);
      return found;
    }
    return null;
  };

  const harvest = async (type) => {
    for (let i = 0; i < 8; i++) {
      const m = await mc.mine_block(type, 64, 1);
      console.log('mined ' + (m.mined || 0) + ' ' + type + ' ' + (m.problem || ''));
      if (m.mined) {
        const c = await mc.collect(type, 64, 64);
        console.log('collected ' + (c.collected || 0) + ' ' + type + ' ' + (c.problem || ''));
        const h = await mc.count(type);
        if (h.count > 0) {
          console.log('got wood: ' + h.count + ' ' + type);
          return true;
        }
      }
    }
    return false;
  };

  const approachAndHarvest = async (type) => {
    const a = await mc.move_to_block(type, 128, 45000);
    console.log('approach ' + type + ' at ' + JSON.stringify(a.at) + ' ' + (a.why || ''));
    if (a.block) {
      await mc.look_at(a.block.x, a.block.y, a.block.z);
      if (await harvest(type)) return true;
    }
    return false;
  };

  const tryKnownLogs = async () => {
    for (const t of logTypes) {
      if (await approachAndHarvest(t)) return true;
    }
    return false;
  };

  if (await tryKnownLogs()) return;

  const found0 = await findLog(160);
  if (found0) {
    if (await approachAndHarvest(found0)) return;
  }

  const step = 6;
  const maxSteps = 40;
  for (let i = 1; i <= maxSteps; i++) {
    const tx = start.x + i * step;
    const tz = start.z;
    const mv = await mc.move_to(tx, start.y, tz, 20000);
    console.log('walk +x step ' + i + ' to ' + JSON.stringify(mv.at) + ' ' + (mv.why || ''));

    const found = await findLog(32);
    if (found) {
      if (await approachAndHarvest(found)) return;
    }

    if (i % 4 === 0) {
      if (await tryKnownLogs()) return;
    }
  }

  throw new Error('no wood found while walking outward');
})();
