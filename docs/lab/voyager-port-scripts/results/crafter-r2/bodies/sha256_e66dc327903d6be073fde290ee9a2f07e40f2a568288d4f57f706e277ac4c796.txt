function ach(){return mc.achievements();}
function num(v){if(typeof v==='number'&&isFinite(v))return Math.max(0,Math.floor(v));if(v===true)return 1;return 0;}
function stoneCount(){return num(ach().collect_stone||0);}
function done(){return num(ach().make_stone_pickaxe||0)>0;}
function tryCraft(){if(done())return true;mc.act('make_stone_pickaxe',1);return done();}
function sweep(){mc.act('do',1);}
function sweepCross(){sweep();mc.act('move_left',1);sweep();mc.act('move_right',1);sweep();mc.act('move_up',1);sweep();mc.act('move_down',1);sweep();}
function sweepSquare(r){for(let dy=-r;dy<=r;dy++){for(let dx=-r;dx<=r;dx++){if(done()||stoneCount()>=3)return;mc.move('right',dx);mc.move('down',dy);sweep();mc.move('left',dx);mc.move('up',dy);}}}
function searchTable(r){for(let dy=-r;dy<=r;dy++){for(let dx=-r;dx<=r;dx++){if(done())return true;mc.move('right',dx);mc.move('down',dy);if(tryCraft())return true;mc.move('left',dx);mc.move('up',dy);}}return false;}
mc.act('sleep',1);
for(let r=0;r<12;r++){if(done()||stoneCount()>=3)break;sweepSquare(r);}
if(!done()&&stoneCount()>=1){for(let r=0;r<16;r++){if(searchTable(r))break;}}
console.log('stone='+stoneCount()+' stone_pickaxe='+done());
