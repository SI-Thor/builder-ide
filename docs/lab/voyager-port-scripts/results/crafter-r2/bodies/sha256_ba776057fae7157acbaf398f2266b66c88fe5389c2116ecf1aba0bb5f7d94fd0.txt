print("sleeping to refill energy")
mc.act("sleep", 1)
print("sweeping nearby tiles for stone")
for r in range(1, 13):
    for dx in range(-r, r + 1):
        for dy in range(-r, r + 1):
            if dx == 0 and dy == 0:
                continue
            if abs(dx) != r and abs(dy) != r:
                continue
            if dx > 0:
                mc.move("right", dx)
            elif dx < 0:
                mc.move("left", -dx)
            if dy > 0:
                mc.move("down", dy)
            elif dy < 0:
                mc.move("up", -dy)
            mc.act("do", 1)
            if dx > 0:
                mc.move("left", dx)
            elif dx < 0:
                mc.move("right", -dx)
            if dy > 0:
                mc.move("up", dy)
            elif dy < 0:
                mc.move("down", -dy)
print("crafting stone pickaxe")
mc.act("make_stone_pickaxe", 1)
print("stone pickaxe crafted: " + str(bool(mc.achievements().get("make_stone_pickaxe"))))
