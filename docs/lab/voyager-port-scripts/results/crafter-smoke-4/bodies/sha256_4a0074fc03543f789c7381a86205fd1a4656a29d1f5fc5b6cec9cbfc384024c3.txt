let energyAction = null;
for (const a of ['energy', 'use_energy', 'consume_energy', 'drink_energy']) {
  try {
    mc.act(a, 1);
    energyAction = a;
    break;
  } catch (e) {}
}

let chopped = false;
let picked = false;
const dirs = ['north', 'south', 'east', 'west'];
const chopActions = ['chop', 'chop_tree', 'hit_tree', 'break_tree', 'mine', 'break', 'hit', 'attack', 'interact', 'use'];
const pickActions = ['pickup', 'pick_up', 'pick', 'take', 'collect', 'gather', 'grab', 'loot', 'interact', 'use'];

for (let r = 0; r <= 6 && !chopped; r++) {
  for (let dx = -r; dx <= r && !chopped; dx++) {
    for (let dy = -r; dy <= r && !chopped; dy++) {
      if (Math.max(Math.abs(dx), Math.abs(dy)) !== r) continue;
      try {
        if (dx > 0) mc.move('east', dx);
        else if (dx < 0) mc.move('west', -dx);
      } catch (e) {}
      try {
        if (dy > 0) mc.move('north', dy);
        else if (dy < 0) mc.move('south', -dy);
      } catch (e) {}

      for (const a of chopActions) {
        try {
          mc.act(a, 1);
          chopped = true;
          break;
        } catch (e) {}
      }

      if (chopped) {
        for (const a of pickActions) {
          try {
            mc.act(a, 1);
            picked = true;
            break;
          } catch (e) {}
        }
      }
    }
  }
}

console.log('energy_action=' + energyAction + ' chopped=' + chopped + ' picked=' + picked);
