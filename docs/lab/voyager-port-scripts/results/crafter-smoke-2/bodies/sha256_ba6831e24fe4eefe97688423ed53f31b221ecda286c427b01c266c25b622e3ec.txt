const energyActions = ["use_energy", "eat_energy", "drink_energy", "consume_energy", "energy"];
let restored = false;
for (const action of energyActions) {
  try {
    const steps = mc.act(action, 1);
    if (steps > 0) {
      restored = true;
      console.log(`Restored energy with ${action}`);
      break;
    }
  } catch (e) {
    // try next known energy action
  }
}
if (!restored) {
  console.log("No energy action succeeded");
}
