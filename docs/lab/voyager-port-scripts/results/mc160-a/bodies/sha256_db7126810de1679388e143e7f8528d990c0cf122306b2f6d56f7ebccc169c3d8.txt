const LOGS=['oak_log','spruce_log','birch_log','jungle_log','acacia_log','dark_oak_log','mangrove_log','cherry_log'];
async function hasWood(){for(const id of LOGS){const r=await mc.has(id);if(r&&r.has)return true;}return false;}
async function collectLogs(){let n=0;for(const id of LOGS){const r=await mc.collect(id,64,64);if(r&&r.collected)n+=r.collected;}return n;}
async function mineLog(id,block){if(!block)return false;const bx=Math.floor(block.x);const by=Math.floor(block.y);const bz=Math.floor(block.z);await mc.look_at(bx,by,bz);for(let i=0;i<960;i++){const m=await mc.mine_block(id,64,1);if(m&&m.mined>0){await collectLogs();if(await hasWood()){console.log('mined '+id+' at '+bx+','+by+','+bz);return true;}}else if(m&&m.problem)break;await mc.sleep(50);}await collectLogs();return await hasWood();}
async function tryMineNearby(){const scan=await mc.scan(160);if(!scan||!scan.nearby)return false;for(const id of LOGS){if(scan.nearby[id]>0){const mv=await mc.move_to_block(id,128,60000);if(mv&&mv.block){if(await mineLog(id,mv.block))return true;}}}return false;}
async function main(){if(await hasWood()){console.log('already have wood');return;}if(await tryMineNearby()){console.log('got wood');return;}let x=64,z=45,y=66;for(let r=8;r<=1024;r+=8){const pts=[[x+r,z],[x-r,z],[x,z+r],[x,z-r],[x+r,z+r],[x+r,z-r],[x-r,z+r],[x-r,z-r]];for(const p of pts){const mv=await mc.move_to(p[0],y,p[1],60000);if(mv&&mv.at){x=p[0];z=p[1];if(typeof mv.at.y==='number')y=mv.at.y;if(await tryMineNearby()){console.log('got wood');return;}}}}throw new Error('no reachable tree found');}
main();
