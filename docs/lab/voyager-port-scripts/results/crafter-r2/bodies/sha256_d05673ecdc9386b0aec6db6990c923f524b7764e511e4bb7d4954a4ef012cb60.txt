print("wood pickaxe: start")

def done():
    return mc.achievements().get("make_wood_pickaxe") is True

def refill():
    for _ in range(12):
        if done():
            return
        mc.act("do", 1)
    for _ in range(4):
        if done():
            return
        mc.act("sleep", 1)

def try_craft():
    if done():
        return True
    mc.act("make_wood_pickaxe", 1)
    return done()

def spiral_search():
    if try_craft():
        return True
    for radius in range(1, 13):
        for dx in range(-radius, radius + 1):
            for dy in range(-radius, radius + 1):
                if max(abs(dx), abs(dy)) != radius:
                    continue
                if try_craft():
                    return True
                if dx != 0:
                    mc.move("right" if dx > 0 else "left", abs(dx))
                if dy != 0:
                    mc.move("down" if dy > 0 else "up", abs(dy))
                if try_craft():
                    return True
                if dy != 0:
                    mc.move("up" if dy > 0 else "down", abs(dy))
                if dx != 0:
                    mc.move("left" if dx > 0 else "right", abs(dx))
        if try_craft():
            return True
    return done()

refill()
if not try_craft():
    spiral_search()
print("wood pickaxe: " + ("crafted" if done() else "failed"))
