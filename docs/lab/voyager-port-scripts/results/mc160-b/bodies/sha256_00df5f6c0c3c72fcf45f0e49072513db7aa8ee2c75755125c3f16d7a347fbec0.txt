const target = { x: 24, y: 77, z: -52 };
const move = mc.move_to_block('oak_log', 16, 30000);
console.log('move_to_block:', move);

let have = 0;
for (let i = 0; i < 8 && have < 1; i++) {
  mc.look_at(target.x, target.y, target.z);
  const mine = mc.mine_block('oak_log', 16, 1);
  console.log('mine attempt', i + 1, mine);
  const collect = mc.collect('oak_log', 1, 16);
  console.log('collect attempt', i + 1, collect);
  const counted = mc.count('oak_log');
  have = counted && typeof counted.count === 'number' ? counted.count : 0;
  if (have < 1) mc.sleep(500);
}
console.log('oak_log count:', have);
