const LOG = 'jungle_log';
const TARGET = { x: 6, y: 64, z: -34 };

function have() {
  const r = mc.count(LOG);
  return r && typeof r.count === 'number' ? r.count : 0;
}

function lookAtTarget() {
  mc.look_at(TARGET.x + 0.5, TARGET.y + 0.5, TARGET.z + 0.5);
  mc.sleep(200);
}

function tryMineAndCollect(maxDistance, mineTries, collectTries) {
  lookAtTarget();
  for (let i = 0; i < mineTries && have() < 1; i++) {
    const m = mc.mine_block(LOG, maxDistance, 1);
    console.log('mine: ' + JSON.stringify(m));
    mc.sleep(250);
  }
  for (let i = 0; i < collectTries && have() < 1; i++) {
    const c = mc.collect(LOG, 1, maxDistance);
    console.log('collect: ' + JSON.stringify(c));
    mc.sleep(200);
  }
  return have();
}

let have = have();
console.log('start ' + LOG + ': ' + have);

if (have < 1) {
  const mv = mc.move_to_block(LOG, 16, 30000);
  console.log('move_to_block: ' + JSON.stringify(mv));
  have = tryMineAndCollect(16, 8, 8);
}

if (have < 1) {
  const spots = [
    [7, 64, -34],
    [5, 64, -34],
    [6, 64, -33],
    [6, 64, -35],
    [7, 65, -34],
    [5, 65, -34],
    [6, 65, -33],
    [6, 65, -35],
    [7, 64, -33],
    [5, 64, -33],
    [7, 64, -35],
    [5, 64, -35]
  ];
  for (const s of spots) {
    if (have >= 1) break;
    const mv = mc.move_to(s[0] + 0.5, s[1] + 0.5, s[2] + 0.5, 15000);
    console.log('move_to ' + s.join(',') + ': ' + JSON.stringify(mv));
    have = tryMineAndCollect(8, 6, 6);
  }
}

if (have < 1) {
  const mv = mc.move_to(TARGET.x + 1.5, TARGET.y + 0.5, TARGET.z + 0.5, 20000);
  console.log('fallback move: ' + JSON.stringify(mv));
  have = tryMineAndCollect(16, 12, 12);
}

console.log('final ' + LOG + ': ' + have);
