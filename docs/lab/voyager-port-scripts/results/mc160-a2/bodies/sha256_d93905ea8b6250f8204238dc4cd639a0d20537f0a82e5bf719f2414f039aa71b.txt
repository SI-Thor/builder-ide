(async () => {
  const target = { x: -3, y: 66, z: 103 };

  const approach = await mc.move_to_block('oak_log', 128, 600000);
  if (approach && approach.block) {
    target.x = approach.block.x;
    target.y = approach.block.y;
    target.z = approach.block.z;
  }

  await mc.look_at(target.x, target.y, target.z);

  let mined = 0;
  for (let i = 0; i < 12; i++) {
    const result = await mc.mine_block('oak_log', 64, 1);
    if (result && result.mined > 0) {
      mined += result.mined;
      break;
    }
    await mc.sleep(500);
  }

  if (mined > 0) {
    await mc.sleep(1000);
    await mc.collect('oak_log', 1, 16);
  }

  const count = mc.count('oak_log').count;
  if (count > 0) {
    console.log('Collected ' + count + ' oak_log.');
    return;
  }

  console.log('Failed to collect oak_log.');
  throw new Error('No oak_log collected.');
})();
