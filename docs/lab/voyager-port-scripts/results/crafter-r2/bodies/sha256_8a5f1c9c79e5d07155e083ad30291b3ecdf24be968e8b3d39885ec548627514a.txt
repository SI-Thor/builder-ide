print("wood pickaxe: start")

def done():
    return mc.achievements().get("make_wood_pickaxe", False) is True

def refill():
    for _ in range(24):
        if done():
            return True
        mc.act("do", 1)
        mc.act("sleep", 1)
    return done()

def try_craft():
    if done():
        return True
    mc.act("make_wood_pickaxe", 1)
    return done()

def search_and_craft():
    if try_craft():
        return True
    for r in range(1, 25):
        for dx in range(-r, r + 1):
            for dy in range(-r, r + 1):
                if max(abs(dx), abs(dy)) != r:
                    continue
                if try_craft():
                    return True
                if dx != 0:
                    mc.move("right" if dx > 0 else "left", abs(dx))
                if dy != 0:
                    mc.move("down" if dy > 0 else "up", abs(dy))
                if try_craft():
                    return True
                if dy != 0:
                    mc.move("up" if dy > 0 else "down", abs(dy))
                if dx != 0:
                    mc.move("left" if dx > 0 else "right", abs(dx))
                if try_craft():
                    return True
    return done()

refill()
if not search_and_craft():
    refill()
    search_and_craft()
print("wood pickaxe: " + ("crafted" if done() else "failed"))
