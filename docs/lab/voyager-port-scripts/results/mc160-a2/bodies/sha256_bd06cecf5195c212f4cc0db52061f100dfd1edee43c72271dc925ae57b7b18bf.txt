(async () => {
  const target = { x: -3, y: 66, z: 103 };
  const center = { x: target.x + 0.5, y: target.y + 0.5, z: target.z + 0.5 };

  const approach = await mc.move_to_block('oak_log', 128, 600000);
  let block = target;
  if (approach && approach.block && typeof approach.block.x === 'number') {
    block = approach.block;
  }

  const spots = [
    { x: block.x + 1, y: block.y, z: block.z },
    { x: block.x - 1, y: block.y, z: block.z },
    { x: block.x, y: block.y, z: block.z + 1 },
    { x: block.x, y: block.y, z: block.z - 1 },
    { x: block.x + 1, y: block.y, z: block.z + 1 },
    { x: block.x - 1, y: block.y, z: block.z - 1 },
    { x: block.x, y: block.y + 1, z: block.z },
    { x: block.x, y: block.y - 1, z: block.z }
  ];

  let mined = 0;
  let lastProblem = 'not attempted';

  for (const spot of spots) {
    if (mined > 0) break;
    const move = await mc.move_to(spot.x, spot.y, spot.z, 60000);
    await mc.look_at(center.x, center.y, center.z);
    await mc.sleep(250);

    for (let i = 0; i < 20 && mined === 0; i++) {
      const r = await mc.mine_block('oak_log', 64, 1);
      if (r && r.mined > 0) {
        mined = r.mined;
      } else {
        lastProblem = (r && r.problem) ? r.problem : 'no oak_log mined';
        await mc.sleep(250);
      }
    }
  }

  if (mined === 0) {
    console.log('Could not break the oak log: ' + lastProblem);
    throw new Error('oak_log not mined');
  }

  await mc.sleep(500);
  let collected = 0;
  for (let i = 0; i < 40 && collected === 0; i++) {
    const r = await mc.collect('oak_log', 64, 64);
    if (r && r.collected > 0) {
      collected = r.collected;
    } else {
      await mc.sleep(250);
    }
  }

  const count = mc.count('oak_log').count;
  if (count > 0) {
    console.log('Got ' + count + ' oak_log.');
    return;
  }

  console.log('Broke the oak log but did not collect it.');
  throw new Error('oak_log not collected');
})();
