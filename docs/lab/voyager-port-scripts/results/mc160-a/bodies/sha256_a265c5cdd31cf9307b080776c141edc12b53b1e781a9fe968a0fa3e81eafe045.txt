const LOGS=['oak_log','spruce_log','birch_log','jungle_log','acacia_log','dark_oak_log','mangrove_log','cherry_log'];
async function hasWood(){for(const id of LOGS){const r=await mc.has(id);if(r&&r.has)return true;}return false;}
async function collectAll(){let n=0;for(const id of LOGS){const r=await mc.collect(id,64,32);if(r&&r.collected)n+=r.collected;}return n;}
async function tryMine(id){const mv=await mc.move_to_block(id,128,30000);if(!mv||!mv.block)return false;const bx=Math.floor(mv.block.x);const by=Math.floor(mv.block.y);const bz=Math.floor(mv.block.z);await mc.look_at(bx,by,bz);for(let i=0;i<24;i++){const m=await mc.mine_block(id,32,1);if(m&&m.mined>0){await collectAll();if(await hasWood()){console.log('mined and collected '+id);return true;}}else if(m&&m.problem)break;}await collectAll();return await hasWood();}
async function main(){if(await hasWood()){console.log('already have wood');return;}let scan=await mc.scan(160);if(scan&&scan.nearby){for(const id of LOGS){if(scan.nearby[id]>0){if(await tryMine(id)){console.log('got wood');return;}}}for(let step=8;step<=256;step+=8){const tx=38+step;const tz=45;const mv=await mc.move_to(tx,67,tz,20000);if(mv&&mv.at){scan=await mc.scan(160);if(scan&&scan.nearby){for(const id of LOGS){if(scan.nearby[id]>0){if(await tryMine(id)){console.log('got wood');return;}}}}}}throw new Error('no reachable tree found');}
main();
