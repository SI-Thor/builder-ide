console.log("start: sleep, go to tree, chop, craft wood pickaxe");

function crafted() {
  return Boolean(mc.achievements().get("make_wood_pickaxe"));
}

if (!crafted()) {
  mc.act("sleep", 1);
  mc.move("up", 2);
  mc.act("do", 12);
  mc.act("make_wood_pickaxe", 1);
}

console.log("wood pickaxe crafted: " + (crafted() ? "yes" : "no"));
