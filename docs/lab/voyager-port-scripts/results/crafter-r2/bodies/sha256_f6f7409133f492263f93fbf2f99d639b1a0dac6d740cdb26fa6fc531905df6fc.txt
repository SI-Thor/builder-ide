function ach(){return mc.achievements();}
function num(v){if(typeof v==='number'&&isFinite(v))return Math.max(0,Math.floor(v));if(v===true)return 1;return 0;}
function stoneCount(){return num(ach().collect_stone||ach()['collect_stone']||0);}
function woodCount(){return num(ach().collect_wood||ach()['collect_wood']||0);}
function done(){const v=ach().make_stone_pickaxe||ach()['make_stone_pickaxe'];return v===true||num(v)>0;}
function sleepIfNeeded(){mc.act('sleep',1);}
function sweepTile(){mc.act('do',1);}
function sweepCross(){sweepTile();mc.act('move_left',1);sweepTile();mc.act('move_right',1);sweepTile();mc.act('move_up',1);sweepTile();mc.act('move_down',1);sweepTile();}
function sweepSquare(r){for(let dy=-r;dy<=r;dy++){for(let dx=-r;dx<=r;dx++){if(done()||stoneCount()>=3)return;mc.move('right',dx);mc.move('down',dy);sweepTile();mc.move('left',dx);mc.move('up',dy);}}}
function chopTree(){mc.move('right',2);mc.move('down',3);mc.act('do',3);mc.move('up',3);mc.move('left',2);}
sleepIfNeeded();
for(let r=0;r<8;r++){if(done()||stoneCount()>=3)break;sweepSquare(r);}
if(woodCount()<2)chopTree();
mc.act('make_stone_pickaxe',1);
console.log('stone='+stoneCount()+' wood='+woodCount()+' stone_pickaxe='+done());
