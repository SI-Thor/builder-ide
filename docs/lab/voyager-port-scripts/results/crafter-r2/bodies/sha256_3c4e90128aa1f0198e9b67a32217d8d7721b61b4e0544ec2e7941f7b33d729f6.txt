console.log("starting wood pickaxe skill");

function done() {
  return !!mc.achievements().make_wood_pickaxe;
}

function woodCount() {
  try {
    return mc.achievements().collect_wood || 0;
  } catch (e) {
    return 0;
  }
}

function tryCraft() {
  if (done()) return true;
  mc.act("place_table", 1);
  mc.act("make_wood_pickaxe", 1);
  return done();
}

for (let i = 0; i < 24 && !done(); i++) {
  mc.act("sleep", 1);
}

const startWood = woodCount();

mc.move("right", 1);
mc.move("down", 1);

for (let i = 0; i < 12 && !done() && woodCount() < startWood + 2; i++) {
  mc.act("do", 1);
}

mc.move("up", 1);
mc.move("left", 1);

for (let i = 0; i < 6 && !done() && woodCount() < startWood + 2; i++) {
  mc.act("do", 1);
}

const crafted = tryCraft();
console.log("wood pickaxe crafted: " + (crafted ? "yes" : "no"));
