const LOGS=['oak_log','spruce_log','birch_log','jungle_log','acacia_log','dark_oak_log','mangrove_log','cherry_log'];
async function hasWood(){for(const id of LOGS){const r=await mc.count(id);if(r&&r.count>0)return true;}return false;}
async function collectLogs(){for(const id of LOGS){const c=await mc.collect(id,64,16);if(c&&c.collected>0)console.log('collected '+c.collected+' '+id);}}
async function tryMine(id){const mv=await mc.move_to_block(id,128,12000);if(mv&&mv.block){await mc.look_at(mv.block.x,mv.block.y,mv.block.z);for(let i=0;i<3;i++){const m=await mc.mine_block(id,8,1);if(!m)break;console.log('mined '+(m.mined||0)+' '+id+' '+(m.problem||''));if(m.mined>0){await collectLogs();if(await hasWood())return true;}}}else{console.log('could not reach '+id+' '+(mv&&mv.why||''));}return false;}
async function main(){if(await hasWood()){console.log('already have wood');return;}let s=await mc.scan(160);let nb=(s&&s.nearby)||{};for(const id of LOGS){if(nb[id]&&await tryMine(id)){console.log('got wood');return;}}const dirs=[[32,0],[0,32],[-32,0],[0,-32]];for(const[dx,dz]of dirs){const mv=await mc.move_to(37.6+dx,67,44.7+dz,8000);if(mv&&mv.at)console.log('moved to '+mv.at.x+','+mv.at.y+','+mv.at.z+' '+(mv.why||''));s=await mc.scan(160);nb=(s&&s.nearby)||{};for(const id of LOGS){if(nb[id]&&await tryMine(id)){console.log('got wood');return;}}}throw new Error('no tree found');}
await main();
