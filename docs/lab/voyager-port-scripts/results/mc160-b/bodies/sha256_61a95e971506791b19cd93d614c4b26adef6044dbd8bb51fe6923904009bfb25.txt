const target = { x: 9, y: 68, z: -39 };
let have = 0;
for (let attempt = 0; attempt < 3 && have < 1; attempt++) {
  const moved = mc.move_to_block('oak_log', 16, 30000);
  console.log('move attempt ' + (attempt + 1) + ': ' + JSON.stringify(moved));
  mc.look_at(target.x, target.y, target.z);
  const mined = mc.mine_block('oak_log', 16, 1);
  console.log('mine attempt ' + (attempt + 1) + ': ' + JSON.stringify(mined));
  mc.sleep(500);
  const collected = mc.collect('oak_log', 1, 64);
  console.log('collect attempt ' + (attempt + 1) + ': ' + JSON.stringify(collected));
  have = mc.count('oak_log').count;
}
console.log('oak_log count: ' + have);
