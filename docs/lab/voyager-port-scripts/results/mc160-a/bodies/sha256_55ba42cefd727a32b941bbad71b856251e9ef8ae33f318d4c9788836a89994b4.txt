const LOGS=['oak_log','spruce_log','birch_log','jungle_log','acacia_log','dark_oak_log','mangrove_log','cherry_log'];
function isLog(n){return typeof n==='string'&&n.endsWith('_log');}
async function hasWood(){for(const id of LOGS){const r=await mc.has(id);if(r&&r.has)return true;}return false;}
async function scanLogIds(radius){const s=await mc.scan(radius);const n=s&&s.nearby?s.nearby:{};const out=[];for(const k of Object.keys(n)){if(n[k]>0&&isLog(k))out.push(k);}return out;}
async function mineOne(id){
  const mv=await mc.move_to_block(id,128,60000);
  if(!mv||!mv.block||typeof mv.block.x!=='number')return false;
  const bx=Math.floor(mv.block.x),by=Math.floor(mv.block.y),bz=Math.floor(mv.block.z);
  await mc.look_at(bx,by,bz);
  for(let i=0;i<120;i++){
    if(await hasWood())return true;
    const m=await mc.mine_block(id,12,1);
    if(m&&m.mined>0){
      const c=await mc.collect(id,64,64);
      if(c&&c.collected>0&&await hasWood())return true;
    }
    await mc.sleep(20);
  }
  return await hasWood();
}
async function tryHere(){
  const found=await scanLogIds(32);
  for(const id of found){if(await mineOne(id))return true;}
  for(const id of LOGS){if(await mineOne(id))return true;}
  return false;
}
async function main(){
  if(await hasWood()){console.log('already have wood');return;}
  let x=59.5,y=67,z=47.7;
  for(let step=0;step<=48;step++){
    if(await tryHere()){console.log('got wood');return;}
    const nx=x+16;
    const mv=await mc.move_to(nx,y,z,30000);
    if(mv&&mv.at&&typeof mv.at.x==='number')x=mv.at.x;
    if(mv&&mv.at&&typeof mv.at.y==='number')y=mv.at.y;
    if(mv&&mv.at&&typeof mv.at.z==='number')z=mv.at.z;
    else x=nx;
  }
  throw new Error('no reachable tree found');
}
main();
