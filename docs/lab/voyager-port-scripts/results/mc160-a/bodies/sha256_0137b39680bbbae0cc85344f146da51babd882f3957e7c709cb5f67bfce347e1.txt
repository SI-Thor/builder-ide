const LOGS=['oak_log','spruce_log','birch_log','jungle_log','acacia_log','dark_oak_log','mangrove_log','cherry_log'];
async function hasWood(){for(const id of LOGS){const r=await mc.has(id);if(r&&r.has)return true;}return false;}
async function mineLog(id){
  const mv=await mc.move_to_block(id,128,60000);
  if(!mv||!mv.block||typeof mv.block.x!=='number')return false;
  const bx=Math.floor(mv.block.x),by=Math.floor(mv.block.y),bz=Math.floor(mv.block.z);
  await mc.look_at(bx,by,bz);
  for(let i=0;i<30;i++){
    if(await hasWood())return true;
    const m=await mc.mine_block(id,8,1);
    if(m&&m.mined>0){
      const c=await mc.collect(id,64,64);
      if(c&&c.collected>0&&await hasWood())return true;
    }
    await mc.sleep(50);
  }
  return await hasWood();
}
async function main(){
  if(await hasWood()){console.log('already have wood');return;}
  let x=63.7,y=66,z=44.3;
  for(let step=0;step<=48;step++){
    if(step===0){
      for(const id of LOGS){if(await mineLog(id)){console.log('got wood');return;}}
      continue;
    }
    const ring=[];
    for(let dx=-step;dx<=step;dx++){
      for(let dz=-step;dz<=step;dz++){
        if(Math.max(Math.abs(dx),Math.abs(dz))===step)ring.push([dx,dz]);
      }
    }
    for(const [dx,dz] of ring){
      const tx=x+dx*8,tz=z+dz*8;
      const mv=await mc.move_to(tx,y,tz,30000);
      if(mv&&mv.at){
        if(typeof mv.at.x==='number')x=mv.at.x;
        if(typeof mv.at.y==='number')y=mv.at.y;
        if(typeof mv.at.z==='number')z=mv.at.z;
      }
      for(const id of LOGS){if(await mineLog(id)){console.log('got wood');return;}}
    }
  }
  throw new Error('no reachable tree found');
}
main();
