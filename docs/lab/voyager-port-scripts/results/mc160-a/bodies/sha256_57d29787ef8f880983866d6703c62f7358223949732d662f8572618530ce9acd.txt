const LOGS=['oak_log','spruce_log','birch_log','jungle_log','acacia_log','dark_oak_log','mangrove_log','cherry_log'];
async function hasWood(){for(const id of LOGS){const r=await mc.has(id);if(r&&r.has)return true;}return false;}
async function collectLogs(){let n=0;for(const id of LOGS){const r=await mc.collect(id,64,64);if(r&&r.collected)n+=r.collected;}return n;}
async function mineAt(block){if(!block)return false;const bx=Math.floor(block.x);const by=Math.floor(block.y);const bz=Math.floor(block.z);await mc.look_at(bx,by,bz);for(let i=0;i<120;i++){const m=await mc.mine_block('oak_log',16,1);if(m&&m.mined>0){await collectLogs();if(await hasWood())return true;}for(const id of LOGS){const m2=await mc.mine_block(id,16,1);if(m2&&m2.mined>0){await collectLogs();if(await hasWood())return true;}}await mc.sleep(20);}await collectLogs();return await hasWood();}
async function tryFindAndMine(){for(const id of LOGS){const mv=await mc.move_to_block(id,128,60000);if(mv&&mv.block){if(await mineAt(mv.block))return true;}}return false;}
async function main(){if(await hasWood()){console.log('already have wood');return;}if(await tryFindAndMine()){console.log('got wood');return;}let x=63,z=44,y=66;for(let r=16;r<=512;r+=16){const pts=[];for(let dx=-r;dx<=r;dx+=16){pts.push([x+dx,z-r]);pts.push([x+dx,z+r]);}for(let dz=-r+16;dz<=r-16;dz+=16){pts.push([x-r,z+dz]);pts.push([x+r,z+dz]);}for(const p of pts){const mv=await mc.move_to(p[0],y,p[1],60000);if(mv&&mv.at){if(typeof mv.at.y==='number')y=mv.at.y;if(await tryFindAndMine()){console.log('got wood');return;}}}}throw new Error('no reachable tree found');}
main();
