const LOG = 'jungle_log';
const TARGET = { x: 5, y: 70, z: -37 };

function countLog() {
  const r = mc.count(LOG);
  return r && typeof r.count === 'number' ? r.count : 0;
}

function tryMineCollect() {
  mc.look_at(TARGET.x + 0.5, TARGET.y + 0.5, TARGET.z + 0.5);
  mc.sleep(200);

  let mined = 0;
  for (let i = 0; i < 16 && mined < 1; i++) {
    const m = mc.mine_block(LOG, 16, 1);
    console.log('mine: ' + JSON.stringify(m));
    if (m && typeof m.mined === 'number') mined += m.mined;
    mc.sleep(250);
  }

  let collected = 0;
  for (let i = 0; i < 16 && collected < 1; i++) {
    const c = mc.collect(LOG, 1, 16);
    console.log('collect: ' + JSON.stringify(c));
    if (c && typeof c.collected === 'number') collected += c.collected;
    mc.sleep(250);
  }

  return countLog();
}

let have = countLog();
console.log('start ' + LOG + ': ' + have);

if (have < 1) {
  const m = mc.move_to_block(LOG, 32, 30000);
  console.log('move_to_block: ' + JSON.stringify(m));
  have = tryMineCollect();
}

if (have < 1) {
  const spots = [
    [5, 70, -36],
    [5, 70, -38],
    [4, 70, -37],
    [6, 70, -37],
    [5, 69, -36],
    [5, 69, -38],
    [4, 69, -37],
    [6, 69, -37],
    [5, 70, -35],
    [5, 70, -39],
    [4, 70, -36],
    [6, 70, -36],
    [4, 70, -38],
    [6, 70, -38],
    [5, 69, -35],
    [5, 69, -39]
  ];

  for (const s of spots) {
    if (have >= 1) break;
    const m = mc.move_to(s[0], s[1], s[2], 12000);
    console.log('move_to ' + s.join(',') + ': ' + JSON.stringify(m));
    have = tryMineCollect();
  }
}

if (have < 1) {
  const m = mc.move_to(TARGET.x, TARGET.y, TARGET.z, 30000);
  console.log('move_to target: ' + JSON.stringify(m));
  have = tryMineCollect();
}

console.log('final ' + LOG + ': ' + have);
