const LOG = 'oak_log';
const TARGET = { x: 0, y: 70, z: -39 };

function haveLog() {
  const r = mc.count(LOG);
  return r && typeof r.count === 'number' && r.count > 0;
}

function tryMineCollect(maxDistance, tries) {
  for (let i = 0; i < tries && !haveLog(); i++) {
    const m = mc.mine_block(LOG, maxDistance, 1);
    console.log('mine: ' + JSON.stringify(m));
    mc.sleep(300);
    const c = mc.collect(LOG, 1, maxDistance);
    console.log('collect: ' + JSON.stringify(c));
    mc.sleep(250);
  }
}

console.log('start ' + LOG + ': ' + (mc.count(LOG) || {}).count);

if (!haveLog()) {
  mc.look_at(TARGET.x + 0.5, TARGET.y + 0.5, TARGET.z + 0.5);
  mc.sleep(200);
  tryMineCollect(16, 3);
}

if (!haveLog()) {
  const mv = mc.move_to_block(LOG, 16, 30000);
  console.log('move_to_block: ' + JSON.stringify(mv));
  const bx = mv && mv.block && typeof mv.block.x === 'number' ? mv.block.x : TARGET.x;
  const by = mv && mv.block && typeof mv.block.y === 'number' ? mv.block.y : TARGET.y;
  const bz = mv && mv.block && typeof mv.block.z === 'number' ? mv.block.z : TARGET.z;
  mc.look_at(bx + 0.5, by + 0.5, bz + 0.5);
  mc.sleep(200);
  tryMineCollect(16, 5);
}

if (!haveLog()) {
  const spots = [
    { x: TARGET.x + 1.5, y: TARGET.y + 0.5, z: TARGET.z + 0.5 },
    { x: TARGET.x - 1.5, y: TARGET.y + 0.5, z: TARGET.z + 0.5 },
    { x: TARGET.x + 0.5, y: TARGET.y + 0.5, z: TARGET.z + 1.5 },
    { x: TARGET.x + 0.5, y: TARGET.y + 0.5, z: TARGET.z - 1.5 },
    { x: TARGET.x + 0.5, y: TARGET.y - 1.5, z: TARGET.z + 0.5 },
    { x: TARGET.x + 0.5, y: TARGET.y + 1.5, z: TARGET.z + 0.5 }
  ];
  for (const s of spots) {
    if (haveLog()) break;
    const mv = mc.move_to(s.x, s.y, s.z, 20000);
    console.log('move_to: ' + JSON.stringify(mv));
    mc.look_at(TARGET.x + 0.5, TARGET.y + 0.5, TARGET.z + 0.5);
    mc.sleep(200);
    tryMineCollect(16, 4);
  }
}

console.log('final ' + LOG + ': ' + (mc.count(LOG) || {}).count);
