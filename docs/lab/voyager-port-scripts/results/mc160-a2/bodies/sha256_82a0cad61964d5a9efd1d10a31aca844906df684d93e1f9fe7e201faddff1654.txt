(async () => {
  const log = 'oak_log';
  const approach = await mc.move_to_block(log, 128, 600000);
  if (!approach || !approach.block) {
    console.log('Could not reach an oak_log.');
    throw new Error('oak_log not reached');
  }

  const block = approach.block;
  const spots = [
    { x: block.x + 1, y: block.y, z: block.z },
    { x: block.x - 1, y: block.y, z: block.z },
    { x: block.x, y: block.y, z: block.z + 1 },
    { x: block.x, y: block.y, z: block.z - 1 },
    { x: block.x + 1, y: block.y, z: block.z + 1 },
    { x: block.x - 1, y: block.y, z: block.z - 1 },
    { x: block.x + 1, y: block.y, z: block.z - 1 },
    { x: block.x - 1, y: block.y, z: block.z + 1 }
  ];

  for (const spot of spots) {
    const move = await mc.move_to(spot.x, spot.y, spot.z, 60000);
    if (move && move.at) {
      const dx = Math.abs(move.at.x - block.x);
      const dy = Math.abs(move.at.y - block.y);
      const dz = Math.abs(move.at.z - block.z);
      if (dx <= 2 && dy <= 2 && dz <= 2) break;
    }
  }

  await mc.look_at(block.x + 0.5, block.y + 0.5, block.z + 0.5);
  await mc.sleep(250);

  let mined = 0;
  for (let i = 0; i < 8 && mined === 0; i++) {
    const result = await mc.mine_block(log, 64, 1);
    if (result && result.mined > 0) mined = result.mined;
    if (mined === 0) await mc.sleep(500);
  }

  if (mined === 0) {
    console.log('Could not break the oak_log.');
    throw new Error('oak_log not mined');
  }

  await mc.sleep(1000);
  let collected = 0;
  for (let i = 0; i < 8 && collected === 0; i++) {
    const result = await mc.collect(log, 1, 64);
    if (result && result.collected > 0) collected = result.collected;
    if (collected === 0) await mc.sleep(500);
  }

  const count = mc.count(log).count;
  if (count > 0) {
    console.log('Got ' + count + ' oak_log.');
    return;
  }

  console.log('Broke oak_log but did not collect it.');
  throw new Error('oak_log not collected');
})();
