const LOGS=['oak_log','spruce_log','birch_log','jungle_log','acacia_log','dark_oak_log','mangrove_log','cherry_log'];
async function hasWood(){for(const id of LOGS){const r=await mc.has(id);if(r&&r.has)return true;}return false;}
async function findLogId(radius){const s=await mc.scan(radius);const n=(s&&s.nearby)||{};for(const id of LOGS){if(n[id]>0)return id;}return null;}
async function tryMine(id){const mv=await mc.move_to_block(id,128,60000);if(!mv||!mv.block||typeof mv.block.x!=='number')return false;const bx=Math.floor(mv.block.x);const by=Math.floor(mv.block.y);const bz=Math.floor(mv.block.z);await mc.look_at(bx,by,bz);for(let i=0;i<240;i++){const m=await mc.mine_block(id,64,64);if(m&&m.mined>0){const c=await mc.collect(id,64,64);if(c&&c.collected>0&&await hasWood())return true;}if(await hasWood())return true;await mc.sleep(25);}return false;}
async function main(){if(await hasWood()){console.log('already have wood');return;}let x=63.7;let y=66;let z=44.3;const id0=await findLogId(128);if(id0&&await tryMine(id0)){console.log('got wood');return;}for(let step=1;step<=24;step++){const targetZ=z-step*4;const mv=await mc.move_to(x,y,targetZ,30000);if(mv&&mv.at){if(typeof mv.at.x==='number')x=mv.at.x;if(typeof mv.at.y==='number')y=mv.at.y;if(typeof mv.at.z==='number')z=mv.at.z;}const id=await findLogId(128);if(id&&await tryMine(id)){console.log('got wood');return;}}throw new Error('no reachable tree found');}
main();
