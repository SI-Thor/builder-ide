const target = { x: 62, y: 103, z: 37 };
const move = mc.move_to(target.x, target.y, target.z, 600000);
if (move.why) console.log('move note: ' + move.why);
mc.look_at(target.x, target.y, target.z);
const mine = mc.mine_block('birch_log', 64, 1);
if (mine.problem) throw new Error(String(mine.problem));
if (mine.mined < 1) throw new Error('birch_log was not mined');
const collect = mc.collect('birch_log', 1, 64);
if (collect.problem) throw new Error(String(collect.problem));
if (collect.collected < 1) throw new Error('birch_log was not collected');
const have = mc.count('birch_log').count;
if (have < 1) throw new Error('birch_log is not in inventory');
console.log('Traveled to birch log, mined ' + mine.mined + ', collected ' + collect.collected + ', have ' + have + '.');
