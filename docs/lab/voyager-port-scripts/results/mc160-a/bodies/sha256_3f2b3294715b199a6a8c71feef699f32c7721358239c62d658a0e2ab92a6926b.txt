const LOGS = ['oak_log','spruce_log','birch_log','jungle_log','acacia_log','dark_oak_log','mangrove_log','cherry_log'];
async function hasWood() {
  for (const id of LOGS) {
    const r = await mc.has(id);
    if (r && r.has) return true;
  }
  return false;
}
async function collectLogs() {
  let n = 0;
  for (const id of LOGS) {
    const r = await mc.collect(id, 64, 64);
    if (r && r.collected) n += r.collected;
  }
  return n;
}
async function mineLog(block) {
  if (!block) return false;
  const bx = Math.floor(block.x);
  const by = Math.floor(block.y);
  const bz = Math.floor(block.z);
  await mc.look_at(bx, by, bz);
  for (let i = 0; i < 24; i++) {
    const m = await mc.mine_block(block, 16, 1);
    if (m && m.mined > 0) {
      await collectLogs();
      if (await hasWood()) return true;
    }
    await mc.sleep(20);
  }
  await collectLogs();
  return await hasWood();
}
async function tryMineNearby() {
  const scan = await mc.scan(64);
  const nearby = (scan && scan.nearby) || {};
  const found = LOGS.filter(id => nearby[id] > 0);
  for (const id of found) {
    const mv = await mc.move_to_block(id, 64, 30000);
    if (mv && mv.block && await mineLog(mv.block)) return true;
  }
  return false;
}
async function main() {
  if (await hasWood()) {
    console.log('already have wood');
    return;
  }
  let x = 63;
  let z = 44;
  let y = 66;
  for (let step = 0; step < 32; step++) {
    const mv = await mc.move_to(x, y, z, 30000);
    if (mv && mv.at) {
      if (typeof mv.at.x === 'number') x = mv.at.x;
      if (typeof mv.at.y === 'number') y = mv.at.y;
      if (typeof mv.at.z === 'number') z = mv.at.z;
    }
    if (await tryMineNearby()) {
      console.log('got wood');
      return;
    }
    x += 16;
  }
  throw new Error('no reachable tree found');
}
main();
