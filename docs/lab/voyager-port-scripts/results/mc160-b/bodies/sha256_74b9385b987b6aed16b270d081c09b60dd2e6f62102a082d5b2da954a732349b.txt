const target = { x: 23, y: 77, z: -52 };
const move = mc.move_to_block('oak_log', 128, 30000);
console.log('moved near oak_log:', move);
mc.look_at(target.x, target.y, target.z);
let collected = 0;
for (let i = 0; i < 5 && collected < 1; i++) {
  const mine = mc.mine_block('oak_log', 16, 1);
  console.log('mine attempt', i + 1, mine);
  mc.sleep(500);
  const collect = mc.collect('oak_log', 1, 16);
  console.log('collect attempt', i + 1, collect);
  collected = mc.count('oak_log').count;
}
console.log('oak_log count:', collected);
