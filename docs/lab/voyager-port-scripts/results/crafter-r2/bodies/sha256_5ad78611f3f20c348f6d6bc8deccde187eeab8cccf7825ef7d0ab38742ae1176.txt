console.log("wood pickaxe: start");
var done = function () {
  return mc.achievements().make_wood_pickaxe === true;
};
var tryCraft = function () {
  if (done()) return true;
  mc.act("make_wood_pickaxe", 1);
  return done();
};
var refill = function () {
  if (done()) return;
  mc.act("do", 8);
  if (done()) return;
  mc.act("sleep", 4);
  if (done()) return;
  mc.act("do", 8);
};
var searchTable = function () {
  var dirs = [
    ["up", 1], ["right", 1], ["down", 1], ["left", 1],
    ["up", 2], ["right", 2], ["down", 2], ["left", 2],
    ["up", 3], ["right", 3], ["down", 3], ["left", 3],
    ["up", 4], ["right", 4], ["down", 4], ["left", 4]
  ];
  for (var i = 0; i < dirs.length && !done(); i++) {
    mc.move(dirs[i][0], dirs[i][1]);
    if (tryCraft()) return true;
  }
  return done();
};
refill();
if (!tryCraft()) {
  searchTable();
}
console.log("wood pickaxe: " + (done() ? "crafted" : "failed"));
