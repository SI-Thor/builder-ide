const LOGS=['oak_log','spruce_log','birch_log','jungle_log','acacia_log','dark_oak_log','mangrove_log','cherry_log','bamboo_block','crimson_stem','warped_stem'];
function hasWood(){for(const t of LOGS){if(mc.has(t).has)return true;}return false;}
function logType(scan){const n=(scan&&scan.nearby)||{};for(const t of LOGS){if(n[t]>0)return t;}return null;}
function harvest(type){
  for(let i=0;i<8;i++){
    const go=mc.move_to_block(type,128,30000);
    if(go&&go.block&&typeof go.block.x==='number'){mc.look_at(go.block.x,go.block.y,go.block.z);}
    const mine=mc.mine_block(type,64,64);
    const collect=mc.collect(type,64,64);
    if((mine&&mine.mined>0)||(collect&&collect.collected>0)){if(hasWood())return true;}
    const near=mc.scan(32);
    if(!logType(near))break;
  }
  return hasWood();
}
if(hasWood()){console.log('Wood already held.');}
else{
  let found=logType(mc.scan(32));
  if(found&&harvest(found)){console.log('Wood collected.');}
  else{
    let x=41.4,y=66,z=1.3;
    let done=false;
    for(let step=1;step<=120&&!done;step++){
      const tx=x+4;
      const mv=mc.move_to(tx,y,z,15000);
      if(mv&&mv.at){x=mv.at.x;y=mv.at.y;z=mv.at.z;}
      const t=logType(mc.scan(32));
      if(t){if(harvest(t)){console.log('Wood collected.');done=true;}}
    }
  }
}
if(hasWood()){console.log('Wood collected.');}else{console.log('No wood found.');throw new Error('No wood found.');}
