const LOG = 'jungle_log';
const TARGET = { x: 11, y: 67, z: -35 };

function countLogs() {
  const r = mc.count(LOG);
  return r && typeof r.count === 'number' ? r.count : 0;
}

function tryMine() {
  mc.look_at(TARGET.x + 0.5, TARGET.y + 0.5, TARGET.z + 0.5);
  mc.sleep(250);

  let mined = 0;
  for (let i = 0; i < 8 && mined < 1; i++) {
    const m = mc.mine_block(LOG, 6, 1);
    console.log('mine: ' + JSON.stringify(m));
    if (m && typeof m.mined === 'number') mined += m.mined;
    mc.sleep(350);
  }

  let collected = 0;
  for (let i = 0; i < 8 && collected < 1; i++) {
    const c = mc.collect(LOG, 1, 8);
    console.log('collect: ' + JSON.stringify(c));
    if (c && typeof c.collected === 'number') collected += c.collected;
    mc.sleep(350);
  }

  return countLogs();
}

let have = countLogs();
console.log('start ' + LOG + ': ' + have);

if (have < 1) {
  const m = mc.move_to_block(LOG, 16, 20000);
  console.log('move_to_block: ' + JSON.stringify(m));
  have = tryMine();
}

if (have < 1) {
  const spots = [
    [10, 66, -35],
    [12, 66, -35],
    [11, 66, -34],
    [11, 66, -36],
    [10, 67, -35],
    [12, 67, -35],
    [11, 67, -34],
    [11, 67, -36]
  ];

  for (const s of spots) {
    if (have >= 1) break;
    const m = mc.move_to(s[0], s[1], s[2], 12000);
    console.log('move_to ' + s.join(',') + ': ' + JSON.stringify(m));
    have = tryMine();
  }
}

console.log('final ' + LOG + ': ' + have);
