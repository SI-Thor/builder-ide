const LOG_BLOCK = 'oak_log';
const LOG_ITEM = 'oak_log';
const TARGET = { x: 9, y: 65, z: -39 };

function logCount() {
  const c = mc.count(LOG_ITEM);
  return c && typeof c.count === 'number' ? c.count : 0;
}

function tryMineAndCollect() {
  mc.look_at(TARGET.x, TARGET.y, TARGET.z);
  mc.sleep(250);
  const mined = mc.mine_block(LOG_BLOCK, 8, 1);
  console.log('mine_block: ' + JSON.stringify(mined));
  mc.sleep(500);
  const collected = mc.collect(LOG_ITEM, 1, 8);
  console.log('collect: ' + JSON.stringify(collected));
  return logCount();
}

let logs = logCount();
console.log('initial oak_log count: ' + logs);

if (logs < 1) {
  const moved = mc.move_to_block(LOG_BLOCK, 16, 30000);
  console.log('move_to_block: ' + JSON.stringify(moved));
  mc.sleep(250);
  logs = tryMineAndCollect();
}

if (logs < 1) {
  const west = mc.move_to(TARGET.x - 1, TARGET.y + 1, TARGET.z, 15000);
  console.log('move_to west: ' + JSON.stringify(west));
  mc.sleep(250);
  logs = tryMineAndCollect();
}

if (logs < 1) {
  const east = mc.move_to(TARGET.x + 1, TARGET.y + 1, TARGET.z, 15000);
  console.log('move_to east: ' + JSON.stringify(east));
  mc.sleep(250);
  logs = tryMineAndCollect();
}

if (logs < 1) {
  const north = mc.move_to(TARGET.x, TARGET.y + 1, TARGET.z - 1, 15000);
  console.log('move_to north: ' + JSON.stringify(north));
  mc.sleep(250);
  logs = tryMineAndCollect();
}

if (logs < 1) {
  const south = mc.move_to(TARGET.x, TARGET.y + 1, TARGET.z + 1, 15000);
  console.log('move_to south: ' + JSON.stringify(south));
  mc.sleep(250);
  logs = tryMineAndCollect();
}

console.log('final oak_log count: ' + logs);
