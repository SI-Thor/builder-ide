console.log("starting wood pickaxe plan");

function done() {
  return !!mc.achievements().make_wood_pickaxe;
}

function tryCraft() {
  if (done()) return true;
  mc.act("make_wood_pickaxe", 1);
  return done();
}

function tryEat() {
  mc.act("do", 1);
  mc.act("sleep", 1);
}

if (!tryCraft()) {
  for (let i = 0; i < 16 && !done(); i++) {
    tryEat();
    if (tryCraft()) break;
  }
}

if (!done()) {
  mc.act("place_table", 1);
  tryCraft();
}

if (!done()) {
  const dirs = ["up", "down", "left", "right"];
  for (let i = 0; i < 8 && !done(); i++) {
    const d = dirs[i % 4];
    mc.move(d, 1);
    mc.act("make_wood_pickaxe", 1);
    if (done()) break;
    mc.move(d === "up" ? "down" : d === "down" ? "up" : d === "left" ? "right" : "left", 1);
  }
}

console.log("wood pickaxe crafted: " + (done() ? "yes" : "no"));
