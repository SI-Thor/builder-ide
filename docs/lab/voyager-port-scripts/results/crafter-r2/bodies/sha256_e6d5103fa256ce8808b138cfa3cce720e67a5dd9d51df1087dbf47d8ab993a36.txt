print("sleeping to refill energy")
mc.act("sleep", 1)

made = lambda: bool(mc.achievements().get("make_stone_pickaxe"))

if not made():
    print("chopping tree 4 left, 2 up")
    mc.move("left", 4)
    mc.move("up", 2)
    mc.act("do", 8)
    mc.move("down", 2)
    mc.move("right", 4)

if not made():
    print("sweeping nearby tiles for stone")
    for r in range(1, 13):
        if made():
            break
        for dx in range(-r, r + 1):
            if made():
                break
            for dy in range(-r, r + 1):
                if made():
                    break
                if dx == 0 and dy == 0:
                    continue
                if abs(dx) != r and abs(dy) != r:
                    continue
                if dx > 0:
                    mc.move("right", dx)
                elif dx < 0:
                    mc.move("left", -dx)
                if dy > 0:
                    mc.move("down", dy)
                elif dy < 0:
                    mc.move("up", -dy)
                mc.act("do", 1)
                if dx > 0:
                    mc.move("left", dx)
                elif dx < 0:
                    mc.move("right", -dx)
                if dy > 0:
                    mc.move("up", dy)
                elif dy < 0:
                    mc.move("down", -dy)

if not made():
    print("searching for a table to craft")
    for r in range(0, 16):
        if made():
            break
        for dx in range(-r, r + 1):
            if made():
                break
            for dy in range(-r, r + 1):
                if made():
                    break
                if dx == 0 and dy == 0:
                    if r == 0:
                        mc.act("make_stone_pickaxe", 1)
                    continue
                if abs(dx) != r and abs(dy) != r:
                    continue
                if dx > 0:
                    mc.move("right", dx)
                elif dx < 0:
                    mc.move("left", -dx)
                if dy > 0:
                    mc.move("down", dy)
                elif dy < 0:
                    mc.move("up", -dy)
                mc.act("make_stone_pickaxe", 1)
                if dx > 0:
                    mc.move("left", dx)
                elif dx < 0:
                    mc.move("right", -dx)
                if dy > 0:
                    mc.move("up", dy)
                elif dy < 0:
                    mc.move("down", -dy)

print("stone pickaxe crafted:", made())
