print("sleeping to refill energy")
mc.act("sleep", 1)

def try_make():
    mc.act("make_stone_pickaxe", 1)
    return bool(mc.achievements().get("make_stone_pickaxe"))

if try_make():
    print("stone pickaxe already available")
else:
    print("collecting wood from nearby trees")
    for r in range(1, 33):
        for dx in range(-r, r + 1):
            for dy in range(-r, r + 1):
                if dx == 0 and dy == 0:
                    continue
                if abs(dx) != r and abs(dy) != r:
                    continue
                if dx > 0:
                    mc.move("right", dx)
                elif dx < 0:
                    mc.move("left", -dx)
                if dy > 0:
                    mc.move("down", dy)
                elif dy < 0:
                    mc.move("up", -dy)
                mc.act("do", 1)
                if dx > 0:
                    mc.move("left", dx)
                elif dx < 0:
                    mc.move("right", -dx)
                if dy > 0:
                    mc.move("up", dy)
                elif dy < 0:
                    mc.move("down", -dy)

    print("sweeping nearby tiles for stone")
    for r in range(1, 49):
        for dx in range(-r, r + 1):
            for dy in range(-r, r + 1):
                if dx == 0 and dy == 0:
                    continue
                if abs(dx) != r and abs(dy) != r:
                    continue
                if dx > 0:
                    mc.move("right", dx)
                elif dx < 0:
                    mc.move("left", -dx)
                if dy > 0:
                    mc.move("down", dy)
                elif dy < 0:
                    mc.move("up", -dy)
                mc.act("do", 1)
                if dx > 0:
                    mc.move("left", dx)
                elif dx < 0:
                    mc.move("right", -dx)
                if dy > 0:
                    mc.move("up", dy)
                elif dy < 0:
                    mc.move("down", -dy)

    print("attempting to craft at current tile")
    if not try_make():
        print("searching nearby tiles for crafting table")
        for r in range(1, 65):
            for dx in range(-r, r + 1):
                for dy in range(-r, r + 1):
                    if dx == 0 and dy == 0:
                        continue
                    if abs(dx) != r and abs(dy) != r:
                        continue
                    if dx > 0:
                        mc.move("right", dx)
                    elif dx < 0:
                        mc.move("left", -dx)
                    if dy > 0:
                        mc.move("down", dy)
                    elif dy < 0:
                        mc.move("up", -dy)
                    if try_make():
                        break
                    if dx > 0:
                        mc.move("left", dx)
                    elif dx < 0:
                        mc.move("right", -dx)
                    if dy > 0:
                        mc.move("up", dy)
                    elif dy < 0:
                        mc.move("down", -dy)
            if try_make():
                break

print("stone pickaxe crafted:", bool(mc.achievements().get("make_stone_pickaxe")))
