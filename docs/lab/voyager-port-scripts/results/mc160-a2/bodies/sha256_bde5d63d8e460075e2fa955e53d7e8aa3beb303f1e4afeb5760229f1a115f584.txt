(async () => {
  const target = { x: -3, y: 66, z: 103 };

  const near = await mc.move_to_block('oak_log', 128, 600000);
  if (near && near.block) {
    target.x = near.block.x;
    target.y = near.block.y;
    target.z = near.block.z;
  }

  const approach = await mc.move_to(target.x, target.y, target.z, 600000);
  if (approach && approach.at) {
    target.x = Math.round(approach.at.x);
    target.y = Math.round(approach.at.y);
    target.z = Math.round(approach.at.z);
  }

  await mc.look_at(target.x + 0.5, target.y + 0.5, target.z + 0.5);
  await mc.sleep(250);

  let mined = 0;
  for (let i = 0; i < 20 && mined === 0; i++) {
    const result = await mc.mine_block('oak_log', 64, 1);
    if (result && result.mined > 0) mined = result.mined;
    if (mined === 0) await mc.sleep(500);
  }

  if (mined === 0) {
    console.log('Could not break the oak log.');
    throw new Error('oak_log not mined');
  }

  await mc.sleep(1000);
  let collected = 0;
  for (let i = 0; i < 20 && collected === 0; i++) {
    const result = await mc.collect('oak_log', 1, 64);
    if (result && result.collected > 0) collected = result.collected;
    if (collected === 0) await mc.sleep(500);
  }

  const count = mc.count('oak_log').count;
  if (count > 0) {
    console.log('Got ' + count + ' oak_log.');
    return;
  }

  console.log('Broke the oak log but did not collect it.');
  throw new Error('oak_log not collected');
})();
