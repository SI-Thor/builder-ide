const LOGS=['oak_log','spruce_log','birch_log','jungle_log','acacia_log','dark_oak_log','mangrove_log','cherry_log'];
async function hasAnyLog(){for(const id of LOGS){const r=await mc.has(id);if(r&&r.has)return true;}return false;}
async function scanLogId(radius){const s=await mc.scan(radius);const n=(s&&s.nearby)||{};for(const id of LOGS){if(n[id]>0)return id;}return null;}
async function mineLog(id){const mv=await mc.move_to_block(id,128,60000);if(!mv||!mv.block||typeof mv.block.x!=='number')return false;const bx=Math.floor(mv.block.x),by=Math.floor(mv.block.y),bz=Math.floor(mv.block.z);await mc.look_at(bx,by,bz);for(let i=0;i<960;i++){const m=await mc.mine_block(id,64,64);if(m&&m.mined>0){const c=await mc.collect(id,64,64);if(c&&c.collected>0&&await hasAnyLog())return true;}if(await hasAnyLog())return true;await mc.sleep(25);}return false;}
async function main(){if(await hasAnyLog()){console.log('already have wood');return;}let x=63.7,y=66,z=44.3;const id0=await scanLogId(128);if(id0&&await mineLog(id0)){console.log('got wood');return;}for(let r=1;r<=12;r++){const pts=[];for(let i=0;i<r*4;i++){const a=i*Math.PI*2/(r*4);pts.push([x+Math.cos(a)*r*8,z+Math.sin(a)*r*8]);}for(const p of pts){const mv=await mc.move_to(p[0],y,p[1],30000);if(mv&&mv.at){if(typeof mv.at.x==='number')x=mv.at.x;if(typeof mv.at.y==='number')y=mv.at.y;if(typeof mv.at.z==='number')z=mv.at.z;}const id=await scanLogId(128);if(id&&await mineLog(id)){console.log('got wood');return;}}}throw new Error('no reachable tree found');}
main();
